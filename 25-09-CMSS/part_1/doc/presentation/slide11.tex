        \frametitle{Ok, but where's the tipping?}

        % Body
        \footnotesize

        \uncover<1->{
                \alr{Remember:}once the \textit{control} parameter is fixed, the long-term behaviour of solutions is set.
        }

        \vspace{1mm}

        \uncover<2->{
                We need somehow to embed some variability in our system. We make our parameter $\mu$ to depend explicitly on time.
        }

        \vspace{1mm}

        \uncover<3->{
                Consider a saddle-node vector field $f$ and a family of parameter shifts $\Lambda_{\varepsilon}$
                \begin{align*}
                        & f(x,\lambda) = -((x+a+b\lambda)^{2}+c \text{tanh}(\lambda - d))\bigg(x - \frac{k}{\text{cosh}(e\lambda)}\bigg)\,,\\
                        & \Lambda_{\varepsilon}(\lambda_{-},\lambda_{+}) = \{\lambda(\varepsilon t)\;:\;\lambda\to\lambda_{\pm},\:t\to\pm\infty\}\,.
                \end{align*}
        }
