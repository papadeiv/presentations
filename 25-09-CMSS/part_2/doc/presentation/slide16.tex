        % Title
        \frametitle{A probabilistic EWS from raw timeseries (?)}

        % Body
        \footnotesize

        \begin{algorithm}[H]
                \KwIn{timeseries $x_{t}$ over $\Delta t > 0$}

                \BlankLine

                define $V_{*}(x,\boldsymbol{c}) = \boldsymbol{c[1]}x + \boldsymbol{c[2]}x^{2} + \boldsymbol{c[3]}x^{3}$\;
                define $\rho(x,\boldsymbol{c})=\exp(-V(x,\boldsymbol{c}))$\;

                \BlankLine
                \BlankLine

                residuals = detrend($x_{t}$, $\Delta t$)\;
                histogram = fit(residuals)\;

                \BlankLine

                \While{$\text{error}<1e-16$}{
                        guess = rand(3)\;
                        $p(x) = \text{normalise}(\rho(x,\text{guess}))$\;
                        error = solve\_nlls($p(x)$, histogram)\;
                }

                \BlankLine

                a, b = compute\_extrema($V_{*}(x, \text{guess})$)\;
                \alr{ews}= $\frac{1}{2\pi}\sqrt{|\pprime{V_{*}}(b, \text{guess})|\pprime{V_{*}}(a, \text{guess})}\exp(-(V_{*}(b, \text{guess})-V_{*}(a, \text{guess})))$\;

                \BlankLine
                \BlankLine

                \KwOut{escape rate\alr{ews}from the minima $a$ of $V_{*}$}
        \end{algorithm}

\end{frame}
