                \uncover<4->{
                        Assuming that the trend is determined by a saddle-node normal form
                        \begin{equation*}
                                \begin{cases}
                                        \dot{x} = f(x,\mu) = -\mu - x^2\,, \\
                                        \mu(t) = \mu_{0} + \varepsilon t\,. 
                                \end{cases}
                        \end{equation*}
                }

                \uncover<5->{
                        The equilibria of the frozen system are
                        \begin{equation*}
                                x_{\text{stb},\text{uns}} = \pm\sqrt{-\mu}\,,\quad\mu\leq0\,.
                        \end{equation*}
                }

                \uncover<6->{
                        Linear stability is prescribed by
                        \begin{equation*}
                                \lambda = \newprime{f}(x=x_{\text{stb},\text{uns}},\mu) = \mp2\sqrt{-\mu}\,.
                        \end{equation*}
                }

       \end{columns}

       \uncover<7->{
                        The saddle-node (or fold) bifurcation at $\mu=0$ causes $x_{\text{stb}}=x_{\text{uns}}$ and for $\mu>0$ no local nor global attractor is left in phase space.
       }
