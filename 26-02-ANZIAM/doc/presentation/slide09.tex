        % Title
        \frametitle{Transition probabilities}

        % Body
        \footnotesize

        \uncover<1->{
                Given a $1-$dimensional conservative system $dX_{t} = -\newprime{V}(X_{t};\mu)dt + \sqrt{2D}dW_{t}$ the transition probability $p(X_{t},t|X_{0},0)$ satifies the Fokker-Plank equation 

                \begin{equation*}
                     \partial_{t}p = \mathcal{L}p := \newprime{(\newprime{V}p)} + D\pprime{p}\,. 
                \end{equation*}
        }

        \vspace{-0.25cm}

        \uncover<2->{
                In the singular limit $\varepsilon\to0$ the transition probability solves the second-order IVP

                \begin{equation*}
                   \begin{cases}
                       D \pprime{p} + \pprime{V}p + \newprime{V}\newprime{p} = 0\,, \\
                       \newprime{p}(\pm\infty) = 0\,.
                   \end{cases}
                \end{equation*}

                whose solution is a\alr{stationary equilibrium density}$p(x) = Z\,\exp(-V(x)/D)$, with $Z>0$ being the normalisation constant of $\exp(-V(x)/D)$.
        }
