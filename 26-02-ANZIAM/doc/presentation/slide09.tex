% Slide # 9 
\begin{frame}[t, label=slide09]
        % Title
        \frametitle{Functional setting}

        % Body
        \footnotesize
        
        Let $I \subset \mathbb{R}$ and $p(x)\in H_{I}:=\big\{f\in L^{2}(I)\;:\;\int_{I}^{}f(x)dx = 1\big\}$.

        \vspace{0.25cm}
        \uncover<2->{
                For any $h>0$ there is a partition $I_{h} = \cup_{k=1}^{M}B_{k}$ of $I$ into homogeneous bins $B_{k}$.
        }

        \vspace{0.25cm}
        \uncover<3->{
                Construct the subspace $\mathcal{H}_{h}:=\big\{f\in H_{I}\;:\;f(x)=\sum_{k=1}^{M}c_{k}\boldsymbol{1}_{B_{k}}(x)\big\}$.
        }

        \vspace{0.35cm}
        \uncover<4->{
                Let $\{X_{n}\}_{n=1}^{N}$ be a sequence of i.i.d. observations sampled from $p(x)$, then 

                \begin{equation*}
                        \hat{p}_{h}(x) = \sum_{k=1}^{M}\hat{p}_{k}\boldsymbol{1}_{B_{k}} = \frac{1}{Nh}\sum_{k=1}^{M}\boldsymbol{1}_{B_{k}}(x)\sum_{n=1}^{N}\boldsymbol{1}_{B_{k}}(X_{n})\,,
                \end{equation*}
        
                is a random function of $\mathcal{H}_{n}$ associated to $\{X_{n}\}_{n=1}^{N}$.
        }

        \vspace{0.0cm}
        \uncover<5->{
                \begin{proposition}
                        The fluctuations between the unbiased density estimator $\hat{p}_{h}$ and the bin-averaged histogram $p_{h}=\sum_{k=1}^{M}p_{k}\boldsymbol{1}_{B_{K}}$, $p_{k}= \frac{1}{h}\int_{B_{k}}^{}p(x)dx$ are asymptotically Gaussian

                        \begin{equation*}
                                \sqrt{N}(\hat{p}_{k}-p_{k})\stackrel{d}{\longrightarrow}\mathcal{N}\bigg(0,\frac{\pi_{k}(1-\pi_{k})}{h^{2}}\bigg)\,,\quad k=1,\dots,M\,,
                        \end{equation*}

                        where $\pi_{k}:=Pr[x\in B_{k}] = h p_{k}$ defines the bin probability. 
                \end{proposition}
        }
\end{frame}
