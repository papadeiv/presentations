        \frametitle{Dynamical systems that tip}

        % Body
        \footnotesize

        Consider $n-$dimensional Itô processes with non-autonomous drift and additive noise

        \begin{equation*}
           \begin{cases}
                   dX_{t} = f(X_{t},\,\mu(t))dt + \sqrt{2D}dW_{t} \,, \\
              \dot{\mu} \in \Lambda_{\varepsilon}(\mu_{-},\mu_{+}) \,. 
           \end{cases}
        \end{equation*}

        \uncover<2->{
                $\Lambda_{\varepsilon}$ is a\alr{parametrised family of shifts} connecting the past-limit (i.e. $\mu(t)\to\mu_{-}$, $t\to-\infty$) to the future-limit system (i.e. $\mu(t)\to\mu_{+}$, $t\to+\infty$).
        }

        \vspace{0.25cm}
        \uncover<3->{
                $X^{*}$ is a\alr{quasi-steady equilibrium}(QSE) of the system if $f(X^{*},\mu(t)) = 0$. 
        }

        \vspace{0.25cm}
        \uncover<4->{
                If a QSE is created, destroyed or changes stability then a\alr{bifurcation}has occured.
        }

        \vspace{0.25cm}
        \uncover<5->{
                Suppose $\Lambda_{\varepsilon}(\mu_{-},\mu_{+}) = \frac{\mu_{-}+\mu_{+}}{2} + \frac{\mu_{+} - \mu_{-}}{2}\tanh(\varepsilon(t-t_{0}))$:
