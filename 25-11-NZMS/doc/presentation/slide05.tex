        % Title
        \frametitle{Critical slowing down (CSD)}

        % Body
        \footnotesize

        Consider the deterministic limit of the noisy saddle-node normal form in $2-$D

        \begin{equation*}
                \begin{cases}
                        \dot{x} = -\mu - x^{2}\,, \\
                        \dot{y} = - \alpha y\,,
                \end{cases}
        \end{equation*}

        with a \textbf{stable node} $\boldsymbol{x}_{\mu}^{(1)} = [\sqrt{-\mu},0]^{T}$ and a \textbf{saddle} $\boldsymbol{x}_{\mu}^{(2)} = [-\sqrt{-\mu},0]^{T}$ equilibria.
 
        \uncover<2->{
                Then (per Hartman-Grobman) the flow close to $\boldsymbol{x}_{\mu}^{(1)}$ is well approximated by

                \begin{equation*}
                \boldsymbol{x}(t) = \begin{bmatrix}
                     x(t) \\
                     y(t)
                \end{bmatrix} =  
                \begin{bmatrix}
                     x_{0} \\
                     y_{0}
                \end{bmatrix} +  
                c_{1}^{(1)}e^{-2\sqrt{-\mu}t}\begin{bmatrix}
                     1 \\
                     0
                \end{bmatrix} + 
                c_{2}^{(1)}e^{-\alpha{t}}\begin{bmatrix}
                     0 \\
                     1
                \end{bmatrix}\,.
                \end{equation*}
        }

        \uncover<3->{
                Let's pick $\alpha = 2\sqrt{-\mu}$ and inject additive white noise with standard deviation $\sigma>0$.
        } 

        \uncover<4->{
                Consider a circular region $\mathcal{B}_{\rho}(\boldsymbol{x}_{\mu}^{(1)})\subset \mathbb{R}^{2}$. How much time until my particles \textit{hit} $\mathcal{B}$?
        }
