% Slide # 8 
\begin{frame}[t, label=slide08]
        % Title
        \frametitle{Transition probabilities}

        % Body
        \footnotesize

        Consider a conservative system $f(x;\mu) = -\nabla V(x;\mu)$ where $V$ is a scalar potential of the state's variables.

        \vspace{2mm}

        \uncover<2->{
                The stochastic evolution of an observable $X_{t}:\Omega\to \mathbb{R}$ under $f$ is in analogy with the motion of an overdamped particle in a potential $V$ subject to noisy diffusion (Langevin equation)

                \begin{equation*}
                        dX_{t} = -\newprime{V}(X_{t};\mu)dt + \sqrt{2D}dW_{t}\,.
                \end{equation*}
        }

        \vspace{1mm}

        \uncover<3->{
                Given $X_{0}\in \mathbb{R}$ the transition probability $p(X_{t},t|X_{0},0)$ is described, over the state space, by the solution of the Fokker-Plank (or forward Kolmogoroff) equation
                \begin{equation*}
                     \partial_{t}p = \mathcal{L}p := \newprime{(\newprime{V}p)} + D\pprime{p}\,. 
                \end{equation*}
        }

        \vspace{1mm}

        \uncover<4->{
                Let $\mu\in \mathbb{R}$ be fixed over $\Delta t >> 0$; if $\frac{\Delta p}{\Delta t}\approx 0$ (i.e. I ignore the transient regime) then my particle's transition probability solves the second-order IVP

                \begin{equation*}
                   \begin{cases}
                       D \pprime{p} + \pprime{V}p + \newprime{V}\newprime{p} = 0\,, \\
                       \newprime{p}(\pm\infty) = 0\,.
                   \end{cases}
                \end{equation*}
        }

\end{frame}
