\begin{frame}[t, label=slide10]
        % Title
        \frametitle{A numerical method to predict tipping}

        % Body
        \footnotesize

        \begin{algorithm}[H]
                \KwIn{timeseries $x_{t}$ over $\Delta t > 0$, threshold $>0$}

                \BlankLine

                define $V(x,\boldsymbol{c}) = \boldsymbol{c[1]}x + \boldsymbol{c[2]}x^{2} + \boldsymbol{c[3]}x^{3}$\;
                define $\rho\,(x,\boldsymbol{c})=\exp(-V(x,\boldsymbol{c}))$\;

                \BlankLine
                \BlankLine

                residuals = detrend($x_{t}$, $\Delta t$)\;
                histogram = fit(residuals)\;

                \BlankLine

                \While{$\text{error}>\text{threshold}$}{
                        guess = solve\_lls(histogram) + rand(3)\;
                        $p(x) = \text{normalise}(\rho\,(x,\text{guess}))$\;
                        solution = solve\_nlls($p(x)$, histogram, guess)\;
                        error = $\int_{}^{}(\rho\,(x,\text{solution}) - \text{histogram})^{2}dx$
                }

                \BlankLine

                a, b = compute\_extrema($V(x, \text{guess})$)\;
                \alr{ews}= $\exp(-(V(b, \text{guess})-V(a, \text{guess})))$\;

                \BlankLine
                \BlankLine

                \KwOut{approximate\alr{ews}from the the reconstruction of $V$}
        \end{algorithm}

\end{frame}
